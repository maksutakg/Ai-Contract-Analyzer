# Authentication (Login & Register) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add an email and password authentication system (Register, Login, Session Validation, Logout) using built-in SQLite and signed HMAC Bearer tokens so that only authenticated users can analyze contracts.

**Architecture:** A zero-external-dependency backend auth module (`src/law_rag/auth.py`) stores users in SQLite (`data/users.db`), hashes passwords with `hashlib.scrypt`, and issues stateless HMAC-SHA256 Bearer tokens. FastAPI endpoints (`/api/v1/auth/register`, `/api/v1/auth/login`, `/api/v1/auth/me`) manage sessions while `POST /api/v1/contracts/analyze` requires a valid Bearer token. The React frontend stores the token in `localStorage`, gates the upload view behind `AuthPanel`, and displays the active user and sign-out button in the top navigation bar.

**Tech Stack:** Python 3.12+, FastAPI, Pydantic v2, `sqlite3`, `hashlib`, `hmac`, React 19, TypeScript, Vite, Vitest, Testing Library.

**Spec:** `docs/superpowers/specs/2026-11-02-auth-design.md`

## Global Constraints

- Zero new third-party Python or npm packages; use Python stdlib (`sqlite3`, `hashlib`, `hmac`, `secrets`, `base64`, `json`, `time`) and existing frontend dependencies.
- Normalize emails with `.strip().lower()`; require valid email format (`@` and domain dot) and minimum password length of 8 characters.
- Error responses from FastAPI must follow the existing shape: `HTTPException(status_code=..., detail={"code": "...", "message": "..."})`.
- LocalStorage key for the Bearer token must be `"law_rag_auth_token"`.
- Always run backend tests with `uv run pytest` and frontend tests with `npm --prefix frontend test`.

```python
import base64
from datetime import datetime, timezone
import hashlib
import hmac
import json
from pathlib import Path
import re
import secrets
import sqlite3
import time

from .models import AuthSessionResponse, UserResponse

EMAIL_PATTERN = re.compile(r"^[^@\s]+@[^@\s]+\.[^@\s]+$")
MIN_PASSWORD_LENGTH = 8


class AuthError(Exception):
    def __init__(self, code: str, message: str, status_code: int) -> None:
        super().__init__(message)
        self.code = code
        self.message = message
        self.status_code = status_code


def _b64url_encode(data: bytes) -> str:
    return base64.urlsafe_b64encode(data).rstrip(b"=").decode("ascii")


def _b64url_decode(value: str) -> bytes:
    padding = "=" * (-len(value) % 4)
    return base64.urlsafe_b64decode((value + padding).encode("ascii"))


def _hash_password(password: str) -> str:
    salt_hex = secrets.token_hex(16)
    derived = hashlib.scrypt(
        password.encode("utf-8"),
        salt=bytes.fromhex(salt_hex),
        n=16384,
        r=8,
        p=1,
    )
    return f"{salt_hex}${derived.hex()}"


def _verify_password(password: str, stored_hash: str) -> bool:
    try:
        salt_hex, hash_hex = stored_hash.split("$", 1)
        derived = hashlib.scrypt(
            password.encode("utf-8"),
            salt=bytes.fromhex(salt_hex),
            n=16384,
            r=8,
            p=1,
        )
    except ValueError:
        return False
    return hmac.compare_digest(derived.hex(), hash_hex)


class AuthStore:
    def __init__(
        self,
        db_path: Path,
        secret_key: str,
        token_ttl_seconds: int = 86400,
    ) -> None:
        self.db_path = Path(db_path)
        self.secret_key = secret_key
        self.token_ttl_seconds = token_ttl_seconds
        self._init_db()

    def _connect(self) -> sqlite3.Connection:
        self.db_path.parent.mkdir(parents=True, exist_ok=True)
        conn = sqlite3.connect(self.db_path)
        conn.row_factory = sqlite3.Row
        return conn

    def _init_db(self) -> None:
        with self._connect() as conn:
            conn.execute(
                """
                CREATE TABLE IF NOT EXISTS users (
                    id INTEGER PRIMARY KEY AUTOINCREMENT,
                    email TEXT UNIQUE NOT NULL,
                    password_hash TEXT NOT NULL,
                    created_at TEXT NOT NULL
                )
                """
            )
            conn.commit()

    def _issue_token(self, user: UserResponse) -> str:
        payload = {
            "sub": user.id,
            "email": user.email,
            "exp": int(time.time()) + self.token_ttl_seconds,
        }
        payload_b64 = _b64url_encode(json.dumps(payload, separators=(",", ":")).encode("utf-8"))
        signature = hmac.new(
            self.secret_key.encode("utf-8"),
            payload_b64.encode("ascii"),
            hashlib.sha256,
        ).digest()
        sig_b64 = _b64url_encode(signature)
        return f"{payload_b64}.{sig_b64}"

    def register_user(self, email: str, password: str) -> AuthSessionResponse:
        normalized_email = email.strip().lower()
        if not EMAIL_PATTERN.match(normalized_email):
            raise AuthError(
                code="invalid_email",
                message="Please enter a valid email address.",
                status_code=400,
            )
        if len(password) < MIN_PASSWORD_LENGTH:
            raise AuthError(
                code="weak_password",
                message=f"Password must be at least {MIN_PASSWORD_LENGTH} characters long.",
                status_code=400,
            )

        password_hash = _hash_password(password)
        created_at = datetime.now(timezone.utc).isoformat()

        try:
            with self._connect() as conn:
                cursor = conn.execute(
                    "INSERT INTO users (email, password_hash, created_at) VALUES (?, ?, ?)",
                    (normalized_email, password_hash, created_at),
                )
                conn.commit()
                user_id = int(cursor.lastrowid)
        except sqlite3.IntegrityError as exc:
            raise AuthError(
                code="email_already_registered",
                message="An account with this email already exists.",
                status_code=409,
            ) from exc

        user = UserResponse(id=user_id, email=normalized_email)
        return AuthSessionResponse(token=self._issue_token(user), user=user)

    def authenticate_user(self, email: str, password: str) -> AuthSessionResponse:
        normalized_email = email.strip().lower()
        with self._connect() as conn:
            row = conn.execute(
                "SELECT id, email, password_hash FROM users WHERE email = ?",
                (normalized_email,),
            ).fetchone()

        if row is None or not _verify_password(password, row["password_hash"]):
            raise AuthError(
                code="invalid_credentials",
                message="Invalid email or password.",
                status_code=401,
            )

        user = UserResponse(id=int(row["id"]), email=str(row["email"]))
        return AuthSessionResponse(token=self._issue_token(user), user=user)

    def verify_token(self, token: str) -> UserResponse:
        try:
            payload_b64, sig_b64 = token.split(".", 1)
            expected_sig = hmac.new(
                self.secret_key.encode("utf-8"),
                payload_b64.encode("ascii"),
                hashlib.sha256,
            ).digest()
            actual_sig = _b64url_decode(sig_b64)
            if not hmac.compare_digest(expected_sig, actual_sig):
                raise ValueError("Signature mismatch")
            payload = json.loads(_b64url_decode(payload_b64).decode("utf-8"))
            user_id = int(payload["sub"])
            exp = int(payload["exp"])
        except Exception as exc:
            raise AuthError(
                code="authentication_required",
                message="Authentication is required. Please sign in.",
                status_code=401,
            ) from exc

        if exp <= int(time.time()):
            raise AuthError(
                code="authentication_required",
                message="Your session has expired. Please sign in again.",
                status_code=401,
            )

        with self._connect() as conn:
            row = conn.execute(
                "SELECT id, email FROM users WHERE id = ?",
                (user_id,),
            ).fetchone()

        if row is None:
            raise AuthError(
                code="authentication_required",
                message="Authentication is required. Please sign in.",
                status_code=401,
            )

        return UserResponse(id=int(row["id"]), email=str(row["email"]))
```

FRONTEND_DIST = Path(__file__).resolve().parents[2] / "frontend" / "dist"


def create_app(
    use_fake_clients: bool = False,
    frontend_dist: Path = FRONTEND_DIST,
    auth_db_path: Path | None = None,
) -> FastAPI:
    app_settings = Settings()
    auth_store = AuthStore(
        db_path=auth_db_path or app_settings.auth_db_path,
        secret_key=app_settings.auth_secret_key,
        token_ttl_seconds=app_settings.auth_token_ttl_seconds,
    )
    app = FastAPI(title="Law RAG MVP")

    def require_current_user(authorization: str | None = Header(default=None)) -> UserResponse:
        if not authorization or not authorization.lower().startswith("bearer "):
            raise HTTPException(

                detail={

            )
        token = authorization[7:].strip()
        try:

export const REGISTER_URL = "/api/v1/auth/register";

  } catch {

export function checkFile(file: File): string | null {
  const name = file.name.toLowerCase();
  if (!ACCEPTED_EXTENSIONS.some((extension) => name.endsWith(extension))) {
    return "Only PDF and DOCX files can be analyzed.";

    return "The file is larger than 10 MB. Choose a smaller file.";

async function submitAuth(url: string, email: string, password: string): Promise<AuthSession> {
  let response: Response;
  try {
    response = await fetch(url, {
      method: "POST",
      headers: { "Content-Type": "application/json" },
      body: JSON.stringify({ email, password }),
    });
  } catch {
    throw new AnalysisError(
      "The analysis server did not respond. Make sure it is running (start.cmd).",
    );
  }
  if (!response.ok) {
    throw new AnalysisError(await errorMessage(response, "Authentication failed"));
  }
  const session = (await response.json()) as AuthSession;
  setStoredToken(session.token);
  return session;
}

export async function registerUser(email: string, password: string): Promise<AuthSession> {
  return submitAuth(REGISTER_URL, email, password);
}

export async function loginUser(email: string, password: string): Promise<AuthSession> {
  return submitAuth(LOGIN_URL, email, password);
}

export async function fetchCurrentUser(): Promise<AuthUser> {
  const token = getStoredToken();
  if (!token) {
    throw new AuthenticationError("Authentication is required. Please sign in.");
  }
  let response: Response;
  try {
    response = await fetch(ME_URL, {
      method: "GET",
      headers: { Authorization: `Bearer ${token}` },
    });
  } catch {
    throw new AnalysisError(
      "The analysis server did not respond. Make sure it is running (start.cmd).",
    );
  }
  if (response.status === 401) {
    clearStoredToken();
    throw new AuthenticationError(await errorMessage(response, "Authentication is required"));
  }
  if (!response.ok) {
    throw new AnalysisError(await errorMessage(response, "Could not verify session"));
  }
  return (await response.json()) as AuthUser;
}

export async function analyzeContract(
  file: File,
  contractType: ContractType | "",
  signal: AbortSignal,
): Promise<ContractReport> {
  const body = new FormData();
  body.append("file", file);
  if (contractType) body.append("contract_type", contractType);

  const token = getStoredToken();
  const headers: Record<string, string> = token ? { Authorization: `Bearer ${token}` } : {};

  let response: Response;
  try {
    response = await fetch(ANALYZE_URL, { method: "POST", body, headers, signal });
  } catch (error) {
    if (signal.aborted) throw error;
    throw new AnalysisError(
      "The analysis server did not respond. Make sure it is running (start.cmd).",
    );
  }
  if (response.status === 401) {
    clearStoredToken();
    throw new AuthenticationError(await errorMessage(response, "Authentication is required"));
  }
  if (!response.ok) throw new AnalysisError(await errorMessage(response));
  return (await response.json()) as ContractReport;
}

async function errorMessage(
  response: Response,
  prefix = "The analysis could not be completed",
): Promise<string> {
  const fallback = `${prefix} (HTTP ${response.status}).`;
  try {
    const { detail } = await response.json();
    if (typeof detail?.message === "string") return detail.message;
    // FastAPI validation errors arrive as a list of {msg, loc}.
    if (Array.isArray(detail) && typeof detail[0]?.msg === "string") return detail[0].msg;
  } catch {
    // Non-JSON body (e.g. a proxy error page): the status line is all we have.
  }
  return fallback;
}

              fill="none"
              stroke="currentColor"
              strokeWidth="2"
              strokeLinecap="round"
              strokeLinejoin="round"
            >
              <path d="M20 6v28M12 34h16M6 12h28M6 12l-4 12M6 12l4 12M34 12l-4 12M34 12l4 12" />
              <path d="M1 24q5 5 10 0zM29 24q5 5 10 0z" fill="currentColor" />
              <circle cx="20" cy="5" r="2" />
            </svg>
            <span className="top-nav-title">Contract Risk Review</span>
          </div>

          <div className="top-nav-actions">
            {user && state.kind === "done" && (
              <button
                type="button"
                className="nav-pill-button"
                onClick={() => setState({ kind: "idle" })}
              >
                Analyze another contract ↗
              </button>
            )}
            {user && (
              <>
                <span className="top-nav-user" title={user.email}>
                  {user.email}
                </span>
                <button type="button" className="button-quiet" onClick={handleSignOut}>
                  Sign out
                </button>
              </>
            )}
          </div>
        </div>
      </header>

      <div className="app">
        <main>
          {authChecking ? (
            <p className="muted">Checking session...</p>
          ) : !user ? (
            <div className="landing">
              <AuthPanel initialError={authError} onAuthenticated={handleAuthenticated} />
              <SourcePlate />
            </div>
          ) : (
            <>
              {state.kind === "idle" && (
                <div className="landing">
                  <UploadPanel error={state.error} onSubmit={start} />
                  <SourcePlate />
                </div>
              )}
              {state.kind === "analyzing" && (
                <WaitingPanel
                  fileName={state.fileName}
                  startedAt={state.startedAt}
                  onCancel={() => controller.current?.abort()}
                />
              )}
              {state.kind === "done" && <Report fileName={state.fileName} report={state.report} />}
            </>
          )}
        </main>
      </div>
    </div>
  );
}
